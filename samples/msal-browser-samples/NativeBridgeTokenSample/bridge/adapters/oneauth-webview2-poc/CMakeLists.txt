cmake_minimum_required(VERSION 3.25)
project(OneAuthWebView2Poc LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This adapter is only for the Windows WebView2 NAA POC.")
endif()
if(NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(ONEAUTH_SOURCE_DIR "" CACHE PATH "Complete direct OneAuth SDK source checkout")
if(ONEAUTH_SOURCE_DIR)
    set(ONEAUTH_TESTS OFF CACHE BOOL "Build OneAuth tests")
    set(ONEAUTH_INTEROP OFF CACHE BOOL "Build shared C/JSON interop")
    set(ONEAUTH_ENABLE_WEBVIEW2 ON CACHE BOOL "Use WebView2 for OneAuth UI")
    set(ONEAUTH_DISABLE_MSHTML ON CACHE BOOL "Disable legacy MSHTML UI")
    add_subdirectory("${ONEAUTH_SOURCE_DIR}" oneauth)
else()
    find_package(OneAuth CONFIG REQUIRED)
endif()
add_library(OneAuthWebView2Poc SHARED adapter.cpp)
target_compile_features(OneAuthWebView2Poc PRIVATE cxx_std_17)
target_compile_definitions(OneAuthWebView2Poc PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN ONEAUTH_POC_EXPORTS)
target_link_libraries(OneAuthWebView2Poc PRIVATE OneAuth::OneAuth)
